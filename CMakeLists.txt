cmake_minimum_required(VERSION 3.24)
project(Tempos VERSION 0.1.6 LANGUAGES C CXX RC)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/resources/update-public-key.txt" TEMPOS_UPDATE_PUBLIC_KEY LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS resources/update-public-key.txt)
if(TEMPOS_UPDATE_PUBLIC_KEY AND NOT TEMPOS_UPDATE_PUBLIC_KEY MATCHES "^[a-fA-F0-9]+$")
  message(FATAL_ERROR "Invalid Tempos update public key")
endif()
string(LENGTH "${TEMPOS_UPDATE_PUBLIC_KEY}" update_key_length)
if(update_key_length AND NOT update_key_length EQUAL 64)
  message(FATAL_ERROR "Ed25519 public key must contain 64 hex digits")
endif()
configure_file(resources/version.h.in generated/version.h @ONLY)
# Ninja does not scan RC includes like C++ includes. Recompile VERSIONINFO when
# the configured version changes, so About and Windows file properties agree.
set_source_files_properties(resources/tempos.rc PROPERTIES
  OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Tempos requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TEMPOS_ASAN "Enable AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus)
if(TEMPOS_ASAN)
  string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO)
endif()
add_library(tempos_update_libs STATIC third_party/monocypher/monocypher.c third_party/monocypher/monocypher-ed25519.c
  third_party/miniz/miniz.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(tempos_update_libs PUBLIC third_party/monocypher third_party/miniz)
target_compile_definitions(tempos_update_libs PUBLIC MINIZ_NO_DEFLATE_APIS MINIZ_NO_ZLIB_APIS MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
add_library(tempos_core STATIC src/model.cpp src/storage.cpp src/providers.cpp src/temperature.cpp src/http.cpp src/weather.cpp src/open_meteo.cpp src/startup.cpp src/calendar.cpp src/location.cpp src/updates.cpp src/update_package.cpp)
target_include_directories(tempos_core PUBLIC src third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(tempos_core PUBLIC tempos_update_libs pdh iphlpapi dxgi gdi32 crypt32 bcrypt winhttp ws2_32 advapi32 shell32 ole32 oleaut32 propsys windowsapp)
set(APP_SOURCES src/main.cpp src/app.cpp src/settings.cpp src/render.cpp src/accessibility.cpp src/update_install.cpp resources/tempos.rc)
add_executable(Tempos WIN32 ${APP_SOURCES})
target_link_libraries(Tempos PRIVATE tempos_core d2d1 dwrite d3d11 dcomp windowscodecs comctl32 uiautomationcore dwmapi wtsapi32 psapi comdlg32 msimg32 uxtheme version)
target_link_options(Tempos PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/tempos.manifest)
add_executable(Tempos.TestHost WIN32 ${APP_SOURCES})
target_compile_definitions(Tempos.TestHost PRIVATE TEMPOS_TEST_HOST=1)
target_link_libraries(Tempos.TestHost PRIVATE version tempos_core d2d1 dwrite d3d11 dcomp windowscodecs comctl32 uiautomationcore dwmapi wtsapi32 psapi comdlg32 msimg32 uxtheme)
target_link_options(Tempos.TestHost PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/tempos.manifest)
include(CTest)
add_executable(Tempos.SensorControl WIN32 src/sensor_control.cpp)
target_link_libraries(Tempos.SensorControl PRIVATE advapi32 shell32 ole32)
target_link_options(Tempos.SensorControl PRIVATE /MANIFEST:EMBED /MANIFESTUAC:NO /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/sensors.manifest)
set_property(TARGET Tempos.SensorControl APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/resources/sensors.manifest)
if(BUILD_TESTING)
  add_executable(Tempos.UpdateCandidate WIN32 ${APP_SOURCES})
  target_compile_definitions(Tempos.UpdateCandidate PRIVATE TEMPOS_TEST_HOST=1 TEMPOS_UPDATE_CANDIDATE=1)
  target_link_libraries(Tempos.UpdateCandidate PRIVATE $<TARGET_PROPERTY:Tempos.TestHost,LINK_LIBRARIES>)
  target_link_options(Tempos.UpdateCandidate PRIVATE $<TARGET_PROPERTY:Tempos.TestHost,LINK_OPTIONS>)
  add_dependencies(Tempos.TestHost Tempos.UpdateCandidate)
  add_executable(tempos_tests tests/core_tests.cpp tests/placement_tests.cpp tests/temperature_tests.cpp tests/weather_tests.cpp tests/api_mock_tests.cpp tests/update_tests.cpp)
  target_link_libraries(tempos_tests PRIVATE tempos_core)
  add_test(NAME core COMMAND tempos_tests)
endif()
install(TARGETS Tempos RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt)
install(FILES third_party/monocypher/LICENCE.md DESTINATION licenses RENAME Monocypher.md)
install(FILES third_party/miniz/LICENSE DESTINATION licenses RENAME miniz.txt)
install(DIRECTORY data DESTINATION .)
add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/data/regions.json ${CMAKE_CURRENT_BINARY_DIR}/data/README.md
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/data
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/data/regions.json ${CMAKE_CURRENT_SOURCE_DIR}/data/README.md ${CMAKE_CURRENT_BINARY_DIR}/data
  DEPENDS data/regions.json data/README.md
  VERBATIM)
add_custom_target(tempos_data DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/data/regions.json ${CMAKE_CURRENT_BINARY_DIR}/data/README.md)
foreach(target Tempos Tempos.TestHost)
  set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/resources/tempos.manifest)
  add_dependencies(${target} tempos_data)
endforeach()
