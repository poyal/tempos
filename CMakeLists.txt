cmake_minimum_required(VERSION 3.24)
project(Tempos VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Tempos requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TEMPOS_ASAN "Enable AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus)
if(TEMPOS_ASAN)
  string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO)
endif()
add_library(tempos_core STATIC src/model.cpp src/storage.cpp src/providers.cpp src/http.cpp src/weather.cpp src/calendar.cpp)
target_include_directories(tempos_core PUBLIC src third_party)
target_link_libraries(tempos_core PUBLIC pdh iphlpapi dxgi crypt32 bcrypt winhttp ws2_32 advapi32 shell32 ole32 oleaut32 propsys)
set(APP_SOURCES src/main.cpp src/app.cpp src/render.cpp src/accessibility.cpp resources/tempos.rc)
add_executable(Tempos WIN32 ${APP_SOURCES})
target_link_libraries(Tempos PRIVATE tempos_core d2d1 dwrite d3d11 dcomp windowscodecs comctl32 uiautomationcore dwmapi wtsapi32 psapi comdlg32)
target_link_options(Tempos PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/tempos.manifest)
add_executable(Tempos.TestHost WIN32 ${APP_SOURCES})
target_compile_definitions(Tempos.TestHost PRIVATE TEMPOS_TEST_HOST=1)
target_link_libraries(Tempos.TestHost PRIVATE tempos_core d2d1 dwrite d3d11 dcomp windowscodecs comctl32 uiautomationcore dwmapi wtsapi32 psapi comdlg32)
target_link_options(Tempos.TestHost PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/tempos.manifest)
include(CTest)
if(BUILD_TESTING)
  add_executable(tempos_tests tests/core_tests.cpp)
  target_link_libraries(tempos_tests PRIVATE tempos_core)
  add_test(NAME core COMMAND tempos_tests)
endif()
install(TARGETS Tempos RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES THIRD_PARTY_NOTICES.md DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt)
install(DIRECTORY data DESTINATION .)
add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/data/regions.json ${CMAKE_CURRENT_BINARY_DIR}/data/README.md
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/data
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/data/regions.json ${CMAKE_CURRENT_SOURCE_DIR}/data/README.md ${CMAKE_CURRENT_BINARY_DIR}/data
  DEPENDS data/regions.json data/README.md
  VERBATIM)
add_custom_target(tempos_data DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/data/regions.json ${CMAKE_CURRENT_BINARY_DIR}/data/README.md)
foreach(target Tempos Tempos.TestHost)
  set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/resources/tempos.manifest)
  add_dependencies(${target} tempos_data)
endforeach()
